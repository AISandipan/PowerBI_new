-- 11. Percentage contribution of each employee to department salary
SELECT Id, FName, Department, 
       ROUND((Salary / SUM(Salary) OVER (PARTITION BY Department)) * 100, 2) AS Percent_Contribution
FROM employee;

-- 12. Employees with the same last name
SELECT * 
FROM employee e1

    SELECT 1 
    FROM employee e2 
    WHERE e1.LName = e2.LName 
      AND e1.Id <> e2.Id
);

-- 13. Employees whose salary is in the top 10% overall
SELECT * 
FROM employee 
WHERE Salary >= (
    SELECT MAX(Salary)
    FROM (
        SELECT Salary
        FROM employee
        ORDER BY Salary DESC
        LIMIT CEIL(SELECT COUNT(*) FROM employee * 0.10)) AS Top10)
);
